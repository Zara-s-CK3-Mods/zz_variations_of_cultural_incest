# Tells other mods (Secrets of Ambrosius) that this mod is active.
on_game_start = {
	on_actions = { zz_valyrian_cultural_incest_set_mod_flag }
}

# Also for saves started before this mod was added
yearly_global_pulse = {
	on_actions = { zz_valyrian_cultural_incest_set_mod_flag }
}

zz_valyrian_cultural_incest_set_mod_flag = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = zz_cultural_incest_mod } }
			set_global_variable = {
				name = zz_cultural_incest_mod
				value = yes
			}
		}
	}
}
